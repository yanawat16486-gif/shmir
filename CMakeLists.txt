cmake_minimum_required(VERSION 3.20)
project(shmir VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(SHMIR_SANITIZE "Enable AddressSanitizer and UBSan" OFF)
if(SHMIR_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
find_library(ARGON2_LIBRARY NAMES argon2 libargon2.so.1 REQUIRED)
add_library(shmir STATIC
 src/core/shamir.c src/crypto/crypto.c src/storage/storage.c src/platform/paths.c)
target_include_directories(shmir PUBLIC include)
target_link_libraries(shmir PUBLIC OpenSSL::Crypto PkgConfig::JSONC ${ARGON2_LIBRARY})
if(UNIX)
 target_compile_definitions(shmir PRIVATE _POSIX_C_SOURCE=200809L)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(shmir PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_executable(shmir-cli src/cli/main.c)
target_link_libraries(shmir-cli PRIVATE shmir)
set_target_properties(shmir-cli PROPERTIES OUTPUT_NAME shmir)
enable_testing()
add_executable(shmir-tests tests/test_shmir.c)
target_link_libraries(shmir-tests PRIVATE shmir)
add_test(NAME unit COMMAND shmir-tests)
add_test(NAME demo COMMAND shmir-cli --demo)
add_test(NAME cli_help COMMAND shmir-cli --help)
option(SHMIR_PYTHON_COMPAT "Enable Python Phase 2 cross-language compatibility test" ON)
if(SHMIR_PYTHON_COMPAT)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME python_compat COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_python_compat.py $<TARGET_FILE:shmir-cli>)
endif()
endif()
